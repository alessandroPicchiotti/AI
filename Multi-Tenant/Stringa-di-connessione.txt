/////////////// Modalita complessa senza specificarlo per ogni entity
public interface IMustHaveTenant
{
    string TenantId { get; set; }
}

public interface ICurrentTenantService
{
    string TenantId { get; }
}

public class ApplicationDbContext : DbContext
{
    private readonly ICurrentTenantService _currentTenantService;

    public ApplicationDbContext(
        DbContextOptions<ApplicationDbContext> options,
        ICurrentTenantService currentTenantService) : base(options)
    {
        _currentTenantService = currentTenantService;
    }

    public DbSet<Order> Orders { get; set; }
    public DbSet<Product> Products { get; set; }

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        base.OnModelCreating(modelBuilder);

        // Applica il filtro a tutte le entità che implementano IMustHaveTenant
        foreach (var entityType in modelBuilder.Model.GetEntityTypes())
        {
            if (typeof(IMustHaveTenant).IsAssignableFrom(entityType.ClrType))
            {
                // Definiamo dinamicamente l'espressione Lambda: e => e.TenantId == _currentTenantService.TenantId
                var parameter = Expression.Parameter(entityType.ClrType, "e");
                var property = Expression.Property(parameter, nameof(IMustHaveTenant.TenantId));
                var tenantIdValue = Expression.Constant(_currentTenantService.TenantId);
                var comparison = Expression.Equal(property, tenantIdValue);
                var lambda = Expression.Lambda(comparison, parameter);

                modelBuilder.Entity(entityType.ClrType).HasQueryFilter(lambda);
            }
        }
    }
}

/////////// Come comportarsi con i salvataggi (SaveChanges)
public override int SaveChanges()
{
    foreach (var entry in ChangeTracker.Entries<IMustHaveTenant>())
    {
        if (entry.State == EntityState.Added)
        {
            entry.Entity.TenantId = _currentTenantService.TenantId;
        }
    }
    return base.SaveChanges();
}

3. (Opzionale) Protezione a livello di Database con la Row-Level Security (RLS)
Se vuoi una sicurezza "di ferro" a prova di errore (nel caso in cui qualcuno dimentichi un IgnoreQueryFilters() nel codice C# o acceda al database direttamente tramite reportistica o strumenti esterni), puoi implementare la Row-Level Security direttamente in SQL Server.

Si compone di due parti: una Funzione di filtro e una Security Policy.
//////////////////////////////

-- 1. Crea una funzione che legge il Tenant corrente dalla sessione di SQL Server
CREATE FUNCTION Security.fn_securitypredicate(@TenantId NVARCHAR(50))
RETURNS TABLE
WITH SCHEMABINDING
AS
    RETURN SELECT 1 AS fn_securitypredicate_result
    WHERE @TenantId = CAST(SESSION_CONTEXT(N'TenantId') AS NVARCHAR(50))
    -- Oppure confrontandolo con uno user di database mappato
    -- WHERE @TenantId = CURRENT_USER; 
GO

-- 2. Applica la policy sulla tabella Orders
CREATE SECURITY POLICY TenantSecurityPolicy
ADD FILTER PREDICATE Security.fn_securitypredicate(TenantId) ON dbo.Orders,
ADD BLOCK PREDICATE Security.fn_securitypredicate(TenantId) ON dbo.Orders
WITH (STATE = ON);


Come si usa dall'applicazione (.NET): Prima di eseguire qualsiasi query o comando sul database, apri la connessione e imposti il contesto di sessione:

SQL
EXEC sp_set_session_context @key=N'TenantId', @value=N'Tenant_A';
A quel punto, anche se un hacker o un bug scrivessero una query senza filtri (SELECT * FROM Orders), SQL Server restituirà solamente i record del Tenant_A filtrandoli a monte nel motore relazionale.
